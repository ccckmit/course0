# 第 11 章：AI 與測試

## 測試的挑戰與 AI 切入點

軟體測試是軟體開發生命週期中不可或缺的一環，但同時也是開發者最常忽略或拖延的活動。傳統測試面臨以下挑戰：

1. **時間成本**：撰寫測試往往需要與撰寫正式程式碼相等的時間
2. **覆蓋率不足**：開發者傾向於測試快樂路徑（happy path），忽略邊界案例
3. **維護負擔**：當正式程式碼變更時，測試程式碼需要同步更新
4. **測試資料準備**：建立真實且全面的測試資料相當耗時
5. **缺乏專業知識**：許多開發者不熟悉如何撰寫高品質的測試

AI 從以下幾個面向切入以解決這些挑戰：

| 測試活動 | AI 切入方式 | 效益 |
|---------|-------------|------|
| 單元測試生成 | 從原始碼推導測試案例 | 大幅減少手動撰寫時間 |
| 測試資料生成 | 生成邊界值、異常值 | 提高測試覆蓋率 |
| 測試維護 | 自動更新斷言、偵測過期測試 | 降低維護成本 |
| 測試優先排序 | 分析變更影響範圍 | 加速 CI 流程 |
| 錯誤偵測 | 靜態分析、模式比對 | 發現被忽略的缺陷 |

## 單元測試生成

### 函數測試生成

AI 可以從函數的簽名、型別註解和 docstring 推導出對應的測試案例。

**原始函數：**

```python
from decimal import Decimal
from typing import Optional

def calculate_shipping_cost(
    weight: float,
    distance: float,
    is_express: bool = False
) -> Decimal:
    """Calculate shipping cost based on weight and distance.

    Args:
        weight: Package weight in kilograms
        distance: Shipping distance in kilometers
        is_express: Whether to use express shipping

    Returns:
        Shipping cost as Decimal

    Raises:
        ValueError: If weight or distance is negative
    """
    if weight < 0 or distance < 0:
        raise ValueError("Weight and distance must be non-negative")

    base_rate = Decimal("5.00")
    weight_cost = Decimal(str(weight)) * Decimal("2.50")
    distance_cost = Decimal(str(distance)) * Decimal("0.10")

    total = base_rate + weight_cost + distance_cost

    if is_express:
        total *= Decimal("1.5")

    return total.quantize(Decimal("0.01"))
```

```python
class TestCalculateShippingCost:
    def test_standard_shipping(self):
        """基本運費計算：重量 2kg，距離 100km"""
        cost = calculate_shipping_cost(weight=2.0, distance=100.0)
        expected = Decimal("5.00") + Decimal("5.00") + Decimal("10.00")
        assert cost == expected.quantize(Decimal("0.01"))

    def test_express_shipping(self):
        """快捷運費：基本運費乘以 1.5"""
        cost = calculate_shipping_cost(weight=1.0, distance=50.0, is_express=True)
        standard = Decimal("5.00") + Decimal("2.50") + Decimal("5.00")
        expected = (standard * Decimal("1.5")).quantize(Decimal("0.01"))
        assert cost == expected

    def test_zero_weight(self):
        """邊界案例：重量為 0"""
        cost = calculate_shipping_cost(weight=0.0, distance=100.0)
        expected = Decimal("5.00") + Decimal("0.00") + Decimal("10.00")
        assert cost == expected.quantize(Decimal("0.01"))

    def test_zero_distance(self):
        """邊界案例：距離為 0"""
        cost = calculate_shipping_cost(weight=5.0, distance=0.0)
        expected = Decimal("5.00") + Decimal("12.50") + Decimal("0.00")
        assert cost == expected.quantize(Decimal("0.01"))

    def test_negative_weight_raises_error(self):
        """錯誤案例：負數重量應拋出 ValueError"""
        with pytest.raises(ValueError, match="must be non-negative"):
            calculate_shipping_cost(weight=-1.0, distance=100.0)

    def test_negative_distance_raises_error(self):
        """錯誤案例：負數距離應拋出 ValueError"""
        with pytest.raises(ValueError, match="must be non-negative"):
            calculate_shipping_cost(weight=1.0, distance=-50.0)

    def test_large_values(self):
        """大數值測試：確保不會溢出"""
        cost = calculate_shipping_cost(weight=1000.0, distance=10000.0)
        assert cost > Decimal("0")

    def test_precision(self):
        """精確度測試：結果應四捨五入至小數點後兩位"""
        cost = calculate_shipping_cost(weight=1.237, distance=50.0)
        assert cost.as_tuple().exponent == -2

    def test_express_doubling_effect(self):
        """比對快捷與標準運費的倍數關係"""
        standard = calculate_shipping_cost(weight=3.0, distance=200.0)
        express = calculate_shipping_cost(weight=3.0, distance=200.0, is_express=True)
        assert express == (standard * Decimal("1.5")).quantize(Decimal("0.01"))

    def test_default_express_false(self):
        """預設參數測試：is_express 預設為 False"""
        cost = calculate_shipping_cost(weight=1.0, distance=1.0)
        cost_express_false = calculate_shipping_cost(
            weight=1.0, distance=1.0, is_express=False
        )
        assert cost == cost_express_false
```

### 邊界案例分析

AI 擅長識別程式碼中的邊界條件並生成對應的測試：

```python
def parse_age(value: str) -> int:
    """Parse age string to integer with validation.

    Valid age range: 0 to 150 inclusive.
    """
    if not value.strip():
        raise ValueError("Age cannot be empty")
    try:
        age = int(value)
    except ValueError:
        raise ValueError(f"Invalid age format: {value}")
    if age < 0 or age > 150:
        raise ValueError(f"Age must be between 0 and 150: {age}")
    return age

# AI 生成的邊界測試
class TestParseAge:
    def test_valid_ages(self):
        """正常年齡範圍"""
        assert parse_age("0") == 0
        assert parse_age("1") == 1
        assert parse_age("150") == 150
        assert parse_age("25") == 25
        assert parse_age("100") == 100

    def test_boundary_negative_one(self):
        """邊界下方：-1 應被拒絕"""
        with pytest.raises(ValueError, match="between 0 and 150"):
            parse_age("-1")

    def test_boundary_151(self):
        """邊界上方：151 應被拒絕"""
        with pytest.raises(ValueError, match="between 0 and 150"):
            parse_age("151")

    def test_boundary_empty_string(self):
        """空字串應被拒絕"""
        with pytest.raises(ValueError, match="empty"):
            parse_age("")

    def test_boundary_whitespace(self):
        """空白字串應被拒絕"""
        with pytest.raises(ValueError, match="empty"):
            parse_age("   ")

    def test_invalid_format(self):
        """非數字格式"""
        with pytest.raises(ValueError, match="Invalid age"):
            parse_age("twenty")
        with pytest.raises(ValueError, match="Invalid age"):
            parse_age("12.5")
        with pytest.raises(ValueError, match="Invalid age"):
            parse_age("0xFF")

    def test_maximum_integer(self):
        """極大值：遠超過範圍"""
        with pytest.raises(ValueError, match="between 0 and 150"):
            parse_age("999999")

    def test_negative_large(self):
        """極小值：遠低於範圍"""
        with pytest.raises(ValueError, match="between 0 and 150"):
            parse_age("-999999")

    def test_leading_zeros(self):
        """前導零：應被正常解析"""
        assert parse_age("007") == 7
        assert parse_age("000") == 0
```

```python
import pytest
from sqlalchemy import create_engine, Column, Integer, String, Float
from sqlalchemy.orm import declarative_base, sessionmaker, Session

Base = declarative_base()

class Product(Base):
    __tablename__ = "products"
    id = Column(Integer, primary_key=True)
    name = Column(String, nullable=False)
    price = Column(Float, nullable=False)
    stock = Column(Integer, default=0)

class ProductRepository:
    def __init__(self, session: Session):
        self.session = session

    def create(self, name: str, price: float, stock: int = 0) -> Product:
        product = Product(name=name, price=price, stock=stock)
        self.session.add(product)
        self.session.commit()
        return product

    def find_by_id(self, product_id: int) -> Product | None:
        return self.session.get(Product, product_id)

    def find_by_name(self, name: str) -> list[Product]:
        return self.session.query(Product).filter(
            Product.name.ilike(f"%{name}%")
        ).all()

    def update_stock(self, product_id: int, new_stock: int) -> Product | None:
        product = self.find_by_id(product_id)
        if product:
            product.stock = new_stock
            self.session.commit()
        return product

    def delete(self, product_id: int) -> bool:
        product = self.find_by_id(product_id)
        if product:
            self.session.delete(product)
            self.session.commit()
            return True
        return False
```

```python
import pytest
from unittest.mock import Mock, patch
from datetime import datetime
import requests

class PaymentGateway:
    def __init__(self, api_key: str):
        self.api_key = api_key
        self.base_url = "https://api.payment.example.com"

    def charge(self, amount: float, currency: str, token: str) -> dict:
        response = requests.post(
            f"{self.base_url}/charges",
            headers={"Authorization": f"Bearer {self.api_key}"},
            json={
                "amount": int(amount * 100),
                "currency": currency,
                "source": token,
            },
            timeout=30,
        )
        response.raise_for_status()
        return response.json()

class OrderService:
    def __init__(self, payment_gateway: PaymentGateway):
        self.payment_gateway = payment_gateway

    def process_payment(self, order_id: str, amount: float, token: str) -> dict:
        if amount <= 0:
            raise ValueError("Amount must be positive")
        result = self.payment_gateway.charge(amount, "usd", token)
        return {
            "order_id": order_id,
            "payment_id": result["id"],
            "status": "completed",
            "amount": amount,
            "processed_at": datetime.now().isoformat(),
        }
```

```python
import pytest
import struct
from typing import Any

def parse_binary_packet(data: bytes) -> dict:
    """Parse a binary network packet.

    Format:
    - 4 bytes: magic number (0xDEADBEEF)
    - 2 bytes: message type
    - 4 bytes: payload length
    - N bytes: payload
    - 2 bytes: checksum
    """
    if len(data) < 12:
        raise ValueError("Packet too short")

    magic = struct.unpack(">I", data[0:4])[0]
    if magic != 0xDEADBEEF:
        raise ValueError(f"Invalid magic number: {hex(magic)}")

    msg_type = struct.unpack(">H", data[4:6])[0]
    payload_length = struct.unpack(">I", data[6:10])[0]

    if len(data) < 12 + payload_length:
        raise ValueError("Packet truncated")

    payload = data[10:10 + payload_length]
    received_checksum = struct.unpack(">H", data[10 + payload_length:12 + payload_length])[0]
    calculated_checksum = sum(data[:-2]) & 0xFFFF

    if received_checksum != calculated_checksum:
        raise ValueError("Checksum mismatch")

    return {
        "type": msg_type,
        "payload": payload,
        "checksum_valid": True,
    }

# AI 生成的結構化模糊測試

            parse_binary_packet(b"")
        with pytest.raises(ValueError, match="too short"):
            parse_binary_packet(b"\x00" * 11)

            parse_binary_packet(b"\x00" * 12)
        with pytest.raises(ValueError, match="magic"):
```

```python
from typing import Optional

@dataclass
class TestResult:
    name: str
    status: str
    duration: float
    error_message: Optional[str] = None

@dataclass
class TestSuiteReport:
    total: int
    passed: int
    failed: int
    skipped: int
    duration: float
    failures: list[TestResult]

def analyze_test_results(results_path: str) -> TestSuiteReport:
    """AI 輔助的測試結果分析。

    讀取 pytest JSON 報告並生成結構化分析結果。
    """
    with open(results_path) as f:
        data = json.load(f)

    tests: list[TestResult] = []
    for report in data.get("report", {}).get("tests", []):
        tests.append(TestResult(
            name=report["nodeid"],
            status=report["outcome"],
            duration=report.get("duration", 0.0),
            error_message=report.get("call", {}).get("longrepr"),

        total=len(tests),
        passed=sum(1 for t in tests if t.status == "passed"),
        failed=len(failed_tests),

        duration=data.get("report", {}).get("duration", 0.0),
        failures=failed_tests,
    )

def generate_improvement_suggestions(report: TestSuiteReport) -> list[str]:
    """AI 根據測試結果提供改善建議。"""
    suggestions = []

    if report.duration > 300:
        suggestions.append("測試執行時間過長，建議將測試分割為多個並行 job")

    if report.failed / report.total > 0.1:
        suggestions.append("失敗率超過 10%，建議優先修復高優先級測試")

    for failure in report.failures:
        if "timeout" in (failure.error_message or "").lower():
            suggestions.append(f"測試 {failure.name} 超時，建議縮短測試範圍或增加逾時時間")

    return suggestions
```

### 自動化測試生成整合

AI 可以在 CI/CD pipeline 中自動生成缺失的測試：

```python
import ast
import inspect
from pathlib import Path
from typing import Any

class AITestGenerator:
    """AI 驅動的測試生成器，整合至 CI/CD 流程。"""

    def __init__(self, source_dir: Path, test_dir: Path) -> None:
        self.source_dir = source_dir
        self.test_dir = test_dir

    def find_untested_functions(self) -> list[dict]:
        """找出尚未被測試覆蓋的函數。"""
        untested: list[dict] = []
        for py_file in self.source_dir.rglob("*.py"):
            if py_file.name.startswith("_"):
                continue
            source_code = py_file.read_text()
            try:
                tree = ast.parse(source_code)
                for node in ast.walk(tree):
                    if isinstance(node, ast.FunctionDef):
                        if not self._has_corresponding_test(node.name):
                            untested.append({
                                "file": str(py_file.relative_to(self.source_dir)),
                                "function": node.name,
                                "line": node.lineno,
                            })
            except SyntaxError:
                continue
        return untested

    def _has_corresponding_test(self, function_name: str) -> bool:
        """檢查是否有對應的測試。"""
        test_prefixes = ["test_", f"test_{function_name}"]
        for test_file in self.test_dir.rglob("test_*.py"):
            content = test_file.read_text()
            for prefix in test_prefixes:
                if f"def {prefix}" in content:
                    return True
        return False

    def generate_test_coverage_report(self) -> dict:
        """生成測試覆蓋率報告。"""
        untested = self.find_untested_functions()
        total_functions = len(untested) + self._count_tested_functions()
        return {
            "total_functions": total_functions,
            "untested_count": len(untested),
            "coverage_percentage": round(
                (total_functions - len(untested)) / total_functions * 100, 2
            ) if total_functions > 0 else 0,
            "untested_functions": untested,
        }

    def _count_tested_functions(self) -> int:
        count = 0
        for test_file in self.test_dir.rglob("test_*.py"):
            content = test_file.read_text()
            tree = ast.parse(content)
            for node in ast.walk(tree):
                if isinstance(node, ast.FunctionDef) and node.name.startswith("test_"):
                    count += 1
        return count
```
