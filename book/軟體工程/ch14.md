# 第14章：Git 與 AI 結合

版本控制是軟體開發的基石，而 Git 已成為業界標準。隨著 AI 技術的進步，將 AI 整合到 Git 工作流程中可以大幅提升開發效率。本章將探討如何利用 AI 輔助 Git 的各個環節，從 Commit 訊息生成到 Code Review、衝突解決以及 CHANGELOG 管理。

## 14.1 AI 輔助 Commit

Commit 是版本控制的基本單位。高品質的 Commit 訊息能幫助團隊理解變更歷史，而 AI 可以在這個過程中提供極大的幫助。

### 14.1.1 Commit 訊息生成

撰寫良好的 Commit 訊息是一項需要經驗的技能。AI 可以分析程式碼變更，自動生成符合團隊規範的 Commit 訊息。

以下是一個使用 OpenAI API 自動生成 Commit 訊息的 Python 工具：

```python
#!/usr/bin/env python3
# ai_commit.py - AI 輔助 Commit 訊息生成工具

import subprocess
import json
import os
import sys
from openai import OpenAI

def get_staged_diff():
    """獲取暫存區的變更內容"""
    result = subprocess.run(
        ["git", "diff", "--cached", "--stat"],
        capture_output=True, text=True
    )
    stat = result.stdout.strip()

    result = subprocess.run(
        ["git", "diff", "--cached"],
        capture_output=True, text=True
    )
    diff = result.stdout

    return stat, diff[:5000]

def get_changed_files():
    """獲取變更的檔案列表"""
    result = subprocess.run(
        ["git", "diff", "--cached", "--name-status"],
        capture_output=True, text=True
    )
    return result.stdout.strip()

def analyze_changes(diff, files):
    """分析變更內容"""
    client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))

    prompt = f"""你是一個專業的軟體工程師，負責分析 Git 變更並生成 Conventional Commits 格式的 Commit 訊息。

變更檔案：
{files}

變更內容：
{diff}

請分析這些變更並輸出 JSON（不要有其他文字）：
{{
  "type": "Commit 類型 (feat/fix/chore/docs/refactor/test/style/perf/ci/build)",
  "scope": "影響的模組或範圍",
  "description": "簡短描述（50 字以內）",
  "body": "詳細說明為何進行此變更以及變更的影響",
  "breaking": false,
  "breaking_description": "如果有破壞性變更，描述變更內容"
}}

類型說明：
- feat: 新功能
- fix: 錯誤修復
- chore: 維護任務
- docs: 文檔更新
- refactor: 重構
- test: 測試相關
- style: 格式變更
- perf: 效能改善
- ci: CI 設定
- build: 建置系統"""

    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],
        response_format={"type": "json_object"},
        temperature=0.3
    )

    return json.loads(response.choices[0].message.content)

def format_commit_msg(analysis):
    """將分析結果格式化為 Commit 訊息"""
    breaking = "!" if analysis.get("breaking") else ""
    scope = f"({analysis['scope']})" if analysis.get("scope") else ""

    lines = [
        f"{analysis['type']}{scope}{breaking}: {analysis['description']}",
        ""
    ]

    if analysis.get("body"):
        lines.append(analysis["body"])
        lines.append("")

    if analysis.get("breaking") and analysis.get("breaking_description"):
        lines.append("BREAKING CHANGE: " + analysis["breaking_description"])
        lines.append("")

    return "\n".join(lines)

def main():
    stat, diff = get_staged_diff()
    if not diff:
        print("沒有暫存的變更。請先使用 git add 暫存變更。")
        sys.exit(1)

    print("正在分析變更...")
    print(f"變更統計：\n{stat}\n")

    files = get_changed_files()
    analysis = analyze_changes(diff, files)

    commit_msg = format_commit_msg(analysis)
    print("=" * 60)
    print("生成的 Commit 訊息：")
    print("=" * 60)
    print(commit_msg)
    print("=" * 60)

    response = input("\n是否使用此 Commit 訊息？(Y/n): ").strip().lower()
    if response in ("", "y", "yes"):
        msg_file = ".git/COMMIT_EDITMSG"
        with open(msg_file, "w") as f:
            f.write(commit_msg)
        subprocess.run(["git", "commit", "--file", msg_file])
        print("Commit 完成！")

if __name__ == "__main__":
    main()
```

### 14.1.2 變更範圍分析

除了生成 Commit 訊息，AI 還可以分析變更的影響範圍，幫助開發者了解此次提交可能影響的系統部分。

```python
# change_impact.py - 變更影響範圍分析

import subprocess
import json
from pathlib import Path

def get_changed_files_detail():
    result = subprocess.run(
        ["git", "diff", "--cached", "--name-only"],
        capture_output=True, text=True
    )
    return [f for f in result.stdout.strip().split('\n') if f]

def analyze_impact(files):
    """根據變更檔案分析影響範圍"""
    impacts = {
        "frontend": [],
        "backend": [],
        "database": [],
        "api": [],
        "config": [],
        "tests": [],
        "docs": []
    }

    for f in files:
        if f.startswith(("src/frontend", "src/web", "static")):
            impacts["frontend"].append(f)
        elif f.startswith(("src/backend", "src/api", "src/service")):
            impacts["backend"].append(f)
        elif f.startswith(("migrations", "db", "schema")):
            impacts["database"].append(f)
        elif "/api/" in f or f.endswith("routes.py") or f.endswith("controller"):
            impacts["api"].append(f)
        elif f.startswith(("config", ".env")):
            impacts["config"].append(f)
        elif f.startswith("test") or "/test" in f:
            impacts["tests"].append(f)
        elif f.startswith(("docs", "README")):
            impacts["docs"].append(f)

    return {k: v for k, v in impacts.items() if v}

def suggest_reviewers(impacts):
    """根據影響範圍建議審查者"""
    reviewers = []
    if "frontend" in impacts:
        reviewers.append("前端團隊")
    if "backend" in impacts:
        reviewers.append("後端團隊")
    if "database" in impacts:
        reviewers.append("資料庫管理員")
    if "api" in impacts:
        reviewers.append("API 團隊")
    return reviewers

def main():
    files = get_changed_files_detail()
    impacts = analyze_impact(files)

    print("## 變更影響分析 ##\n")
    print(f"變更檔案數：{len(files)}")
    print(f"變更檔案：{', '.join(files)}\n")

    print("影響範圍：")
    for area, affected_files in impacts.items():
        print(f"  [{area}]")
        for f in affected_files:
            print(f"    - {f}")

    reviewers = suggest_reviewers(impacts)
    if reviewers:
        print(f"\n建議審查者：{', '.join(reviewers)}")

if __name__ == "__main__":
    main()
```

### 14.1.3 Conventional Commits 分類

Conventional Commits 是一種 Commit 訊息規範，它定義了結構化的提交訊息格式。AI 可以自動判斷變更類型並進行分類。

```python
# conventional_classifier.py

import re
from enum import Enum

class CommitType(Enum):
    FEAT = "feat"
    FIX = "fix"
    CHORE = "chore"
    DOCS = "docs"
    REFACTOR = "refactor"
    TEST = "test"
    STYLE = "style"
    PERF = "perf"
    CI = "ci"
    BUILD = "build"

def classify_by_diff(diff):
    """根據 diff 內容自動分類 Commit 類型"""
    type_scores = {t.value: 0 for t in CommitType}

    keywords = {
        "feat": ["add", "implement", "new", "feature", "introduce", "support"],
        "fix": ["fix", "bug", "error", "crash", "issue", "incorrect", "wrong"],
        "refactor": ["refactor", "extract", "rename", "restructure", "simplify"],
        "docs": ["doc", "comment", "readme", "documentation", "example"],
        "test": ["test", "spec", "assert", "mock", "coverage"],
        "perf": ["performance", "optimize", "speed", "fast", "slow", "latency"],
        "style": ["format", "lint", "prettier", "eslint", "indent"],
    }

    diff_lower = diff.lower()

    for commit_type, words in keywords.items():
        for word in words:
            if word in diff_lower:
                type_scores[commit_type] += 1

    added_files = re.findall(r'^\+\+\+\s+b/(.+)', diff, re.MULTILINE)
    for f in added_files:
        if f.startswith("test"):
            type_scores["test"] += 2
        elif f.startswith("docs"):
            type_scores["docs"] += 2
        elif f.endswith((".md", ".txt")):
            type_scores["docs"] += 1

    breaking_patterns = [
        r'BREAKING\s*CHANGE',
        r'removed?\s+(function|class|api|endpoint)',
        r'delete\s+(function|class|api|endpoint)'
    ]
    is_breaking = any(re.search(p, diff, re.IGNORECASE) for p in breaking_patterns)

    best_type = max(type_scores, key=type_scores.get)
    if type_scores[best_type] == 0:
        best_type = "chore"

    return best_type, is_breaking
```

## 14.2 AI 輔助 Code Review

Code Review 是保證程式碼品質的關鍵環節。AI 可以協助 Reviewer 更有效率地進行審查。

### 14.2.1 PR 摘要生成

當開發者提交 Pull Request 時，AI 可以自動生成 PR 摘要，幫助 Reviewer 快速了解變更內容。

```python
#!/usr/bin/env python3
# pr_summary.py - 自動生成 PR 摘要

import subprocess
import json
import os
from openai import OpenAI

def get_pr_diff(base_branch="main"):
    result = subprocess.run(
        ["git", "diff", base_branch, "HEAD"],
        capture_output=True, text=True
    )
    return result.stdout

def get_commit_messages(base_branch="main"):
    result = subprocess.run(
        ["git", "log", f"{base_branch}..HEAD", "--oneline"],
        capture_output=True, text=True
    )
    return result.stdout

def generate_pr_summary(diff, commits):
    client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))

    prompt = f"""分析以下 Pull Request 的變更內容，生成一個結構化的 PR 摘要。

Commits:
{commits}

Diff（前 4000 字元）:
{diff[:4000]}

請輸出 JSON 格式的摘要：
{{
  "title": "PR 標題",
  "overview": "變更概述（2-3 句話）",
  "changes": [
    {{"file": "檔案路徑", "summary": "變更摘要"}}
  ],
  "areas_affected": ["影響的系統模組"],
  "testing_notes": "測試建議",
  "reviewer_focus": "審查重點"
}}"""

    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],
        response_format={"type": "json_object"},
        temperature=0.3
    )

    return json.loads(response.choices[0].message.content)

def format_pr_summary(summary):
    output = []
    output.append(f"# {summary['title']}")
    output.append("")
    output.append("## 概述")
    output.append(summary["overview"])
    output.append("")
    output.append("## 變更內容")
    for change in summary.get("changes", []):
        output.append(f"- **{change['file']}**：{change['summary']}")
    output.append("")
    output.append("## 影響範圍")
    for area in summary.get("areas_affected", []):
        output.append(f"- {area}")
    output.append("")
    output.append("## 審查重點")
    output.append(summary.get("reviewer_focus", ""))
    output.append("")
    output.append("## 測試建議")
    output.append(summary.get("testing_notes", ""))

    return "\n".join(output)

def main():
    base_branch = os.getenv("BASE_BRANCH", "main")
    print(f"正在分析從 {base_branch} 到 HEAD 的變更...\n")

    diff = get_pr_diff(base_branch)
    commits = get_commit_messages(base_branch)

    if not diff:
        print("沒有發現變更。")
        return

    summary = generate_pr_summary(diff, commits)
    output = format_pr_summary(summary)
    print(output)

if __name__ == "__main__":
    main()
```

# impact_analysis.py

import ast
import subprocess

def analyze_python_imports(file_path):
    """分析 Python 檔案的匯入關係"""
    with open(file_path, 'r') as f:
        try:
            tree = ast.parse(f.read())
        except SyntaxError:
            return []

    imports = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                imports.append(alias.name)
        elif isinstance(node, ast.ImportFrom):
            module = node.module or ""
            for alias in node.names:
                imports.append(f"{module}.{alias.name}")
    return imports

def find_affected_tests(changed_files):
    """根據變更檔案找出相關的測試"""
    affected_tests = []
    for f in changed_files:
        test_file = f.replace("src/", "tests/")
        test_file = test_file.replace(".py", "_test.py")
        result = subprocess.run(
            ["git", "ls-files", test_file],
            capture_output=True, text=True
        )
        if result.stdout.strip():
            affected_tests.append(test_file)
    return affected_tests

def suggest_test_cases(changed_files):
    suggestions = []
    for f in changed_files:

## 14.3 合併衝突解決

合併衝突是團隊協作中常見的問題。AI 可以幫助分析衝突原因並提供解決方案。

### 14.3.1 衝突原因分析

當發生合併衝突時，AI 可以分析衝突的上下文，幫助開發者理解衝突的根源。

```python
#!/usr/bin/env python3
# conflict_analyzer.py

import subprocess
import re
import json
from openai import OpenAI

def get_conflict_files():
    result = subprocess.run(
        ["git", "diff", "--name-only", "--diff-filter=U"],
        capture_output=True, text=True
    )
    return [f for f in result.stdout.strip().split('\n') if f]

def parse_conflict_markers(content):
    """解析衝突標記"""
    pattern = r'<<<<<<<\s*(.+?)\n(.*?)=======\n(.*?)>>>>>>>\s*(.+?)(?:\n|$)'
    matches = list(re.finditer(pattern, content, re.DOTALL))

    conflicts = []
    for m in matches:
        conflicts.append({
            "ours_label": m.group(1).strip(),
            "ours": m.group(2).strip(),
            "theirs": m.group(3).strip(),
            "theirs_label": m.group(4).strip()
        })
    return conflicts

def analyze_conflict(file_path, conflict_section):
    client = OpenAI()

    prompt = f"""分析以下 Git 合併衝突並提供解決建議。

檔案：{file_path}

當前分支（ours）的內容：
```python
{conflict_section['ours']}
```

合併進來的分支（theirs）的內容：
```python
{conflict_section['theirs']}
```

請分析：
1. 衝突的原因
2. 兩邊程式碼的意圖差異
3. 建議的解決方案
4. 是否兩邊都需要保留

輸出 JSON：
{{
  "cause": "衝突原因分析",
  "intent_ours": "當前分支的修改意圖",
  "intent_theirs": "合併分支的修改意圖",
  "resolution_strategy": "建議的解決策略（ours/theirs/manual/merge）",
  "suggested_resolution": "建議的解決方案程式碼",
  "notes": "其他注意事項"
}}"""

    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],
        response_format={"type": "json_object"}
    )

    return json.loads(response.choices[0].message.content)

def resolve_auto_mergeable(ours, theirs):
    """嘗試自動合併明顯不相關的變更"""
    if ours.startswith(theirs) or theirs.startswith(ours):
        return max([ours, theirs], key=len)

    our_functions = set(re.findall(r'def (\w+)', ours))
    their_functions = set(re.findall(r'def (\w+)', theirs))

    if our_functions.isdisjoint(their_functions):
        return ours + '\n' + theirs

    return None

def main():
    conflict_files = get_conflict_files()
    if not conflict_files:
        print("沒有發現合併衝突。")
        return

    print(f"發現 {len(conflict_files)} 個衝突檔案：")
    for f in conflict_files:
        print(f"  - {f}")
    print()

    for file_path in conflict_files:
        print(f"=== 分析衝突檔案：{file_path} ===")

        with open(file_path, 'r') as f:
            content = f.read()

        conflicts = parse_conflict_markers(content)
        print(f"發現 {len(conflicts)} 個衝突區域\n")

        for i, conflict in enumerate(conflicts, 1):
            print(f"--- 衝突區域 {i} ---")

            merged = resolve_auto_mergeable(conflict["ours"], conflict["theirs"])
            if merged:
                print("此衝突可由 AI 自動合併\n")
            else:
                print("需要 AI 輔助分析...")
                analysis = analyze_conflict(file_path, conflict)
                print(f"衝突原因：{analysis['cause']}")
                print(f"解決策略：{analysis['resolution_strategy']}")
                print(f"建議方案：{analysis['suggested_resolution'][:200]}...\n")

if __name__ == "__main__":
    main()
```

## 14.4 CHANGELOG 生成

CHANGELOG 是專案的重要文件，記錄了每個版本的變更。AI 可以自動從 Commit 歷史中生成結構化的 CHANGELOG。

### 14.4.1 從 Commit 歷史生成

```python
#!/usr/bin/env python3
# changelog_gen.py - CHANGELOG 自動生成工具

import subprocess
import re
import json
from datetime import datetime
from collections import defaultdict

def get_commits_since(tag=None):
    """獲取從指定 tag 以來的所有 Commit"""
    cmd = ["git", "log", "--pretty=format:%H|%an|%ae|%ad|%s|%b", "--date=short"]
    if tag:
        cmd.insert(2, f"{tag}..HEAD")

    result = subprocess.run(cmd, capture_output=True, text=True)
    commits = []

    for line in result.stdout.strip().split('\n'):
        if not line:
            continue
        parts = line.split('|', 5)
        if len(parts) >= 6:
            hash_val, author, email, date, subject, body = parts
            commits.append({
                "hash": hash_val,
                "author": author,
                "email": email,
                "date": date,
                "subject": subject,
                "body": body
            })

    return commits

def parse_conventional(subject):
    """解析 Conventional Commits 格式"""
    pattern = r'^(\w+)(\((.+)\))?(!)?\s*:\s*(.*)'
    match = re.match(pattern, subject)

    if not match:
        return {"type": "other", "scope": None, "breaking": False, "description": subject}

    return {
        "type": match.group(1),
        "scope": match.group(3),
        "breaking": match.group(4) == "!",
        "description": match.group(5)
    }

def categorize_commits(commits):
    """將 Commits 分類"""
    categories = {
        "feat": [],
        "fix": [],
        "docs": [],
        "refactor": [],
        "perf": [],
        "test": [],
        "chore": [],
        "other": []
    }

    breaking_changes = []

    for commit in commits:
        parsed = parse_conventional(commit["subject"])
        commit.update(parsed)

        category = parsed["type"] if parsed["type"] in categories else "other"
        categories[category].append(commit)

        if parsed["breaking"] or "BREAKING CHANGE" in commit.get("body", ""):
            breaking_changes.append(commit)

    return categories, breaking_changes

def generate_changelog(categories, breaking_changes, version, date):
    """生成 CHANGELOG 內容"""
    lines = []

    lines.append(f"## [{version}] - {date}")
    lines.append("")

    if breaking_changes:
        lines.append("### 破壞性變更")
        lines.append("")
        for commit in breaking_changes:
            body = commit.get("body", "").replace("BREAKING CHANGE: ", "")
            lines.append(f"- {body or commit['description']}")
            lines.append("")

    if categories["feat"]:
        lines.append("### 新功能")
        lines.append("")
        for commit in categories["feat"]:
            scope = f"**{commit['scope']}**: " if commit['scope'] else ""
            lines.append(f"- {scope}{commit['description']} ({commit['hash'][:7]})")
        lines.append("")

    if categories["fix"]:
        lines.append("### Bug 修復")
        lines.append("")
        for commit in categories["fix"]:
            scope = f"**{commit['scope']}**: " if commit['scope'] else ""
            lines.append(f"- {scope}{commit['description']} ({commit['hash'][:7]})")
        lines.append("")

    category_labels = {
        "docs": "文檔更新",
        "refactor": "重構",
        "perf": "效能改善",
        "test": "測試",
        "chore": "維護"
    }

    for cat, label in category_labels.items():
        if categories[cat]:
            lines.append(f"### {label}")
            lines.append("")
            for commit in categories[cat]:
                scope = f"**{commit['scope']}**: " if commit['scope'] else ""
                lines.append(f"- {scope}{commit['description']} ({commit['hash'][:7]})")
            lines.append("")

    return "\n".join(lines)

def main():
    import argparse

    parser = argparse.ArgumentParser(description="CHANGELOG 生成工具")
    parser.add_argument("--version", required=True, help="版本號")
    parser.add_argument("--since", help="從哪個 tag 開始")
    parser.add_argument("--output", default="CHANGELOG.md", help="輸出檔案")

    args = parser.parse_args()

    print("正在分析 Commit 歷史...")
    commits = get_commits_since(args.since)
    categories, breaking = categorize_commits(commits)

    date = datetime.now().strftime("%Y-%m-%d")
    changelog = generate_changelog(categories, breaking, args.version, date)

    with open(args.output, 'w') as f:
        f.write(changelog)

    print(f"CHANGELOG 已寫入 {args.output}")
    print(changelog[:500])

if __name__ == "__main__":
    main()
```

# breaking_change_detector.py

import ast
import subprocess

class BreakingChangeDetector(ast.NodeVisitor):
    """使用 AST 分析檢測破壞性變更"""

    def __init__(self):
        self.removed_functions = []
        self.removed_classes = []
        self.changed_signatures = []
        self.removed_parameters = []
        self.current_functions = {}

    def visit_FunctionDef(self, node):
        params = [arg.arg for arg in node.args.args]
        self.current_functions[node.name] = {
            "params": params,
            "lineno": node.lineno
        }
        self.generic_visit(node)

    def compare_with_base(self, base_functions):
        """與基礎版本的函數進行比較"""
        breaking = []
        for name, info in self.current_functions.items():
            if name in base_functions:
                base_params = base_functions[name]["params"]
                if info["params"] != base_params:
                    breaking.append({
                        "type": "signature_change",
                        "function": name,
                        "old_params": base_params,
                        "new_params": info["params"],
                        "description": f"函數 {name} 的參數已變更"
                    })
        return breaking

def extract_public_api(src_dir):
    """提取公開 API"""
    functions = {}
    for path in subprocess.run(
        ["find", src_dir, "-name", "*.py"],
        capture_output=True, text=True
    ).stdout.strip().split('\n'):
        if not path:
            continue
        with open(path, 'r') as f:
            try:
                tree = ast.parse(f.read())
                detector = BreakingChangeDetector()
                detector.visit(tree)
                functions.update(detector.current_functions)
            except SyntaxError:
                continue
    return functions

### 14.6.5 最佳實踐總結

使用 AI 輔助 Git 工作流程時，應遵循以下原則：

1. **漸進式導入**：先從 Commit 訊息生成開始，逐步導入 Code Review 和衝突解決
2. **人工審核**：AI 的建議僅供參考，最終決定權仍應由開發者掌握
3. **持續改進**：根據團隊回饋持續調整 AI 模型和規則
4. **隱私考量**：確保程式碼內容不會透過 API 呼叫外洩，可考慮使用本地 LLM
5. **客製化規則**：根據團隊的編碼規範和習慣客製化 AI 提示詞
6. **失敗處理**：AI 服務不可用時應有降級機制，不阻擋開發流程
7. **漸進檢查**：在 pre-commit 中只做輕量檢查，完整檢查留給 CI 階段

## 總結

本章介紹了如何將 AI 整合到 Git 工作流程的各個環節。從 Commit 訊息生成、Code Review、衝突解決到 CHANGELOG 管理，AI 可以大幅提升開發效率。透過本章的程式碼範例，讀者可以立即開始在自己的專案中實作 AI-assisted Git Workflow。關鍵在於找到 AI 輔助與人工判斷之間的最佳平衡點，讓 AI 處理重複性工作，而開發者專注於需要深入理解的決策。
