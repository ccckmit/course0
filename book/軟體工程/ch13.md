# 第十三章：AI 與維護

## 13.1 概述

軟體維護佔據了軟體開發生命週期中超過 70% 的成本與時間。傳統的維護工作依賴資深工程師的經驗來診斷問題、調校效能、管理技術債。然而，隨著系統規模日益龐大與複雜，純靠人力維護已經變得不可行。AI 技術的引入正在從根本上改變軟體維護的運作方式。

本章將探討 AI 在軟體維護各面向的應用，包括 AI 輔助除錯、效能調校、技術債管理、異常偵測與預測性維護，以及依賴管理等關鍵領域。透過實際的程式碼範例與工具實作，幫助讀者建立 AI 驅動的維護思維與實戰能力。

## 13.2 AI 輔助除錯

### 13.2.1 錯誤訊息分析

除錯是軟體維護中最耗時也最需要經驗的活動。AI 可以從多個層面協助開發者加速除錯過程。首先是錯誤訊息的分析與解讀。現代的錯誤訊息通常包含豐富的上下文資訊，但開發者往往需要花費大量時間在搜尋引擎上尋找相關解答。透過自然語言處理技術，AI 可以即時解析錯誤訊息，並根據歷史問題資料庫提供精確的診斷建議。

以下是 AI 輔助錯誤分析的實作範例：

```python
"""
AI 輔助錯誤分析工具
解析錯誤訊息、分類問題類型、提供修復建議
"""

import json

    def analyze(self, error_message: str) -> Optional[ErrorAnalysis]:
        for name, config in self.error_patterns.items():
            match = re.search(config["pattern"], error_message, re.IGNORECASE)
            if match:

                    code_example=self._get_code_example(name),
                    related_patterns=self._find_related(name),
                )
        return ErrorAnalysis(
            error_type="unknown",
            severity="unknown",
            root_cause="無法自動識別錯誤類型，請提供更多上下文資訊",
            affected_component="unknown",
            suggested_fixes=["搜尋錯誤訊息中的關鍵字以找到相關解決方案"],
        )

            "index_error": "存取序列的索引超出範圍，可能是空序列或邊界條件錯誤",
            "value_error": "數值轉換失敗，輸入資料格式不正確",
            "timeout": "外部服務回應逾時，可能是服務負載過高或網路問題",

            "out_of_memory": "應用程式使用的記憶體超過了可用上限，可能需要增加資源或檢查記憶體洩漏",

    def _identify_component(self, message: str) -> str:
        file_match = re.search(r'(?:File\s+"([^"]+)"|at\s+(\S+\.\w+):\d+)', message)
        if file_match:
            return file_match.group(1) or file_match.group(2)

        if module_match:
            return module_match.group(1)

            ],
            "attribute_error": [

                "確認 API 版本是否相容",
                "使用 dir() 或 Object.keys() 檢查可用屬性",
                "考慮使用 hasattr() 或 in 運算子先行檢查",
            ],
            "type_error": [

                "使用 isinstance() 檢查輸入資料型別",
                "檢查函式的參數順序是否正確",
                "確認外部 API 的回傳格式是否變更",
            ],
            "key_error": [

            "value_error": [
                "檢查輸入字串格式是否符合預期",

            ],
            "timeout": [

                "檢查外部服務的狀態頁面",

                "檢查防火牆規則與安全群組",
                "確認連線埠號是否正確",
                "檢查服務發現與 DNS 解析",

                "檢查是否有記憶體洩漏",
                "使用記憶體分析器（profiler）找出瓶頸",

            ],
            "null_pointer": [

    def _get_code_example(self, error_type: str) -> Optional[str]:
        examples = {
            "key_error": (
                "# 不安全的寫法\n"
                "value = my_dict['key']  # 可能拋出 KeyError\n\n"
                "# 安全的寫法\n"
                "value = my_dict.get('key', default_value)\n\n"
                "# 或使用 try/except\n"
                "try:\n"
                "    value = my_dict['key']\n"
                "except KeyError:\n"
                "    value = default_value\n"
            ),
            "null_pointer": (
                "# 不安全的寫法\n"
                "result = obj.get_value()  # obj 可能為 None\n\n"
                "# 安全的寫法\n"
                "if obj is not None:\n"
                "    result = obj.get_value()\n"
                "else:\n"
                "    result = default_value\n\n"
                "# 或使用 guard clause\n"
                "if not obj:\n"
                "    return default_value\n"
                "result = obj.get_value()\n"
            ),
        }
        return examples.get(error_type)

    def _find_related(self, error_type: str) -> List[str]:
        related_map = {
            "module_not_found": ["import_error", "dependency_conflict"],
            "attribute_error": ["type_error", "api_mismatch"],
            "null_pointer": ["attribute_error", "type_error"],
            "timeout": ["connection_refused", "service_unavailable"],
        }
        return related_map.get(error_type, [])

    def batch_analyze(self, error_logs: List[str]) -> List[ErrorAnalysis]:
        results = []
        category_counts = {}
        for log in error_logs:
            analysis = self.analyze(log)
            if analysis:
                results.append(analysis)
                cat = analysis.error_type
                category_counts[cat] = category_counts.get(cat, 0) + 1
        print("錯誤分類統計:")
        for cat, count in sorted(category_counts.items(), key=lambda x: -x[1]):
            print(f"  {cat}: {count} 次")
        return results

    if analysis.code_example:
        print(f"程式碼範例:\n{analysis.code_example}")
```

### 13.2.2 堆疊追蹤解讀

堆疊追蹤（Stack Trace）是除錯時最重要的線索來源，但對於經驗不足的開發者來說，長串的呼叫鏈往往難以解讀。AI 可以自動解析堆疊追蹤，找出真正的問題根源，並過濾掉無關的框架程式碼。

以下是 AI 輔助的堆疊追蹤分析工具：

```python
"""
AI 堆疊追蹤解析器
自動提取關鍵資訊並過濾框架噪聲
"""
import re
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass


@dataclass
class StackFrame:
    file: str
    line: int
    function: str
    module: str
    is_application_code: bool


@dataclass
class StackTraceAnalysis:
    exception_type: str
    exception_message: str
    frames: List[StackFrame]
    root_cause_frame: Optional[StackFrame]
    application_frames: List[StackFrame]
    framework_frames: List[StackFrame]
    summary: str


class StackTraceParser:
    def __init__(self):
        self.framework_paths = [
            "node_modules/", "site-packages/", "lib/python",
            "vendor/", ".gradle/", "target/classes",
            "java/lang/", "javax/", "org/springframework",
            "django/", "flask/", "express/", "fastapi/",
            "sqlalchemy/", "celery/", "pandas/",
        ]
        self.known_frameworks = [
            "django", "flask", "fastapi", "express", "spring",
            "sqlalchemy", "celery", "pandas", "numpy", "tensorflow",
            "pytorch", "react", "vue", "angular",
        ]

    def parse(self, trace_text: str) -> Optional[StackTraceAnalysis]:
        lines = trace_text.strip().split("\n")
        if not lines:
            return None
        exception_info = self._parse_exception(lines[0])
        frames = []
        for line in lines[1:]:
            frame = self._parse_frame(line)
            if frame:
                frames.append(frame)
        app_frames = [f for f in frames if f.is_application_code]
        framework_frames = [f for f in frames if not f.is_application_code]
        root_cause = app_frames[-1] if app_frames else (frames[-1] if frames else None)
        summary = self._generate_summary(exception_info, root_cause, app_frames)
        return StackTraceAnalysis(
            exception_type=exception_info[0],
            exception_message=exception_info[1],
            frames=frames,
            root_cause_frame=root_cause,
            application_frames=app_frames,
            framework_frames=framework_frames,
            summary=summary,
        )

    def _parse_exception(self, line: str) -> Tuple[str, str]:
        match = re.match(r"^(\w+(?:Error|Exception)):\s*(.*)", line)
        if match:
            return match.group(1), match.group(2)
        match = re.match(r"^Traceback.*", line)
        if match:
            return "Exception", line
        return "Unknown", line

    def _parse_frame(self, line: str) -> Optional[StackFrame]:
        patterns = [
            r'File\s+"([^"]+)",\s+line\s+(\d+),\s+in\s+(\w+)',
            r'\s+at\s+(\S+)\.(\w+)\((\S+):(\d+)\)',
            r'\s+at\s+(\S+)\.(\w+)\(.*\)',
        ]
        for pattern in patterns:
            match = re.search(pattern, line)
            if match:
                groups = match.groups()
                if len(groups) == 3:
                    filepath, line_no, func = groups
                    module = self._extract_module(filepath)
                elif len(groups) == 4:
                    filepath = groups[2]
                    line_no = groups[3]
                    func = groups[1]
                    module = groups[0]
                else:
                    continue
                return StackFrame(
                    file=filepath,
                    line=int(line_no),
                    function=func,
                    module=module,
                    is_application_code=not self._is_framework(filepath, module),
                )
        return None

    def _extract_module(self, filepath: str) -> str:
        parts = filepath.replace("\\", "/").split("/")
        for i, part in enumerate(parts):
            if part.endswith(".py") or part.endswith(".js") or part.endswith(".java"):
                return part.replace(".py", "").replace(".js", "").replace(".java", "")
        return parts[-1] if parts else "unknown"

    def _is_framework(self, filepath: str, module: str) -> bool:
        for path in self.framework_paths:
            if path in filepath:
                return True
        for fw in self.known_frameworks:
            if fw in module.lower() or fw in filepath.lower():
                return True
        return False

    def _generate_summary(
        self,
        exception: Tuple[str, str],
        root_cause: Optional[StackFrame],
        app_frames: List[StackFrame],
    ) -> str:
        ex_type, ex_msg = exception
        if root_cause:
            location = f"{root_cause.file}:{root_cause.line} 的 {root_cause.function}()"
            summary = f"在 {location} 發生 {ex_type}: {ex_msg}"
        else:
            summary = f"發生 {ex_type}: {ex_msg}"
        if len(app_frames) > 1:
            call_chain = " -> ".join(
                f"{f.function}()" for f in app_frames[:3]
            )
            if len(app_frames) > 3:
                call_chain += f" -> ... (共 {len(app_frames)} 層)"
            summary += f"\n呼叫鏈: {call_chain}"
        return summary


parser = StackTraceParser()
sample_trace = """TypeError: unsupported operand type(s) for +: 'int' and 'str'
  File "/app/services/order_service.py", line 45, in calculate_total
    return sum(item.price for item in items) + tax_rate
  File "/app/models/order.py", line 23, in price
    return self.base_price + self.tax_amount
  File "/usr/lib/python3.11/enum.py", line 456, in __getattr__
    raise AttributeError(name) from None
  File "/app/utils/formatting.py", line 12, in format_currency
    return f"${value:.2f}"
"""

analysis = parser.parse(sample_trace)
if analysis:
    print(f"例外類型: {analysis.exception_type}")
    print(f"例外訊息: {analysis.exception_message}")
    print(f"\n摘要: {analysis.summary}")
    print(f"\n應用程式呼叫鏈:")
    for frame in analysis.application_frames:
        marker = " <-- 根本原因" if frame == analysis.root_cause_frame else ""
        print(f"  {frame.file}:{frame.line} {frame.function}(){marker}")
    print(f"\n框架呼叫層數: {len(analysis.framework_frames)}")
```

### 13.2.3 根本原因分析

根本原因分析（Root Cause Analysis, RCA）是除錯過程中最關鍵也最困難的步驟。AI 可以透過比對錯誤模式、分析變更歷史與系統狀態，自動推斷問題的根本原因。

```python
"""
AI 根本原因分析工具
整合錯誤資訊、程式碼變更與系統指標進行 RCA
"""
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
from datetime import datetime, timedelta
from collections import Counter


@dataclass
class CodeChange:
    file: str
    author: str
    commit_hash: str
    timestamp: datetime
    description: str
    lines_added: int
    lines_removed: int


@dataclass
class SystemState:
    cpu_usage: float
    memory_usage: float
    request_rate: float
    error_rate: float
    active_users: int


@dataclass
class RCAResult:
    root_cause: str
    confidence: float
    evidence: List[str]
    suggested_actions: List[str]
    related_changes: List[CodeChange]


class RootCauseAnalyzer:
    def __init__(self):
        self.change_history: List[CodeChange] = []
        self.known_patterns: Dict[str, Dict] = {
            "database_slowdown": {
                "indicators": ["query timeout", "connection pool exhausted", "slow query"],
                "likely_causes": ["缺少索引", "大量資料掃描", "鎖競爭", "連線池不足"],
                "fixes": ["檢查慢查詢日誌", "使用 EXPLAIN ANALYZE 分析查詢計畫", "增加適當的索引"],
            },
            "memory_leak": {
                "indicators": ["out of memory", "heap size growing", "GC overhead"],
                "likely_causes": ["未釋放的物件參考", "快取無上限", "執行緒洩漏"],
                "fixes": ["使用 heap dump 分析物件存活情況", "檢查靜態集合的使用", "實作弱參考"],
            },
            "api_regression": {
                "indicators": ["endpoint failed", "status code 500", "response changed"],
                "likely_causes": ["API 合約變更", "序列化/反序列化錯誤", "驗證邏輯變更"],
                "fixes": ["比對 OpenAPI 規範", "檢查最近的 API 相關提交", "執行合約測試"],
            },
        }

    def add_code_change(self, change: CodeChange):
        self.change_history.append(change)

    def analyze(
        self,
        error_type: str,
        error_message: str,
        current_state: SystemState,
        time_window: timedelta = timedelta(hours=24),
    ) -> RCAResult:
        evidence = []
        recent_changes = [
            c for c in self.change_history
            if c.timestamp > datetime.now() - time_window
        ]
        evidence.append(f"最近 {time_window.days * 24} 小時內有 {len(recent_changes)} 個程式碼變更")
        matched_pattern = None
        for pattern_name, pattern_info in self.known_patterns.items():
            for indicator in pattern_info["indicators"]:
                if indicator.lower() in error_message.lower():
                    matched_pattern = pattern_name
                    evidence.append(f"錯誤訊息符合已知模式: {pattern_name}")
                    break
        if current_state.error_rate > 5:
            evidence.append(f"錯誤率異常偏高: {current_state.error_rate}%")
        if current_state.cpu_usage > 90:
            evidence.append(f"CPU 使用率過高: {current_state.cpu_usage}%")
        if current_state.memory_usage > 90:
            evidence.append(f"記憶體使用率過高: {current_state.memory_usage}%")
        recent_files = Counter(c.file for c in recent_changes)
        top_files = recent_files.most_common(5)
        if top_files:
            evidence.append(f"變更最頻繁的檔案: {', '.join(f[0] for f in top_files)}")
        root_cause, confidence, actions = self._determine_root_cause(
            matched_pattern, recent_changes, current_state, error_message
        )
        return RCAResult(
            root_cause=root_cause,
            confidence=confidence,
            evidence=evidence,
            suggested_actions=actions,
            related_changes=recent_changes[:5],
        )

    def _determine_root_cause(
        self,
        pattern: Optional[str],
        changes: List[CodeChange],
        state: SystemState,
        error: str,
    ) -> Tuple[str, float, List[str]]:
        if pattern and pattern in self.known_patterns:
            info = self.known_patterns[pattern]
            return info["likely_causes"][0], 0.7, info["fixes"]
        for change in changes:
            if "migration" in change.file.lower() or ".sql" in change.file.lower():
                return (
                    f"資料庫 Migration 可能導致問題: {change.description}",
                    0.6,
                    ["檢查 Migration 腳本", "執行資料庫回滾測試", "驗證資料完整性"],
                )
            if "config" in change.file.lower() or ".env" in change.file.lower():
                return (
                    f"配置變更可能導致問題: {change.description}",
                    0.5,
                    ["還原配置變更", "檢查配置值是否正確", "比對不同環境的配置差異"],
                )
        if state.memory_usage > 90:
            return "記憶體不足導致服務不穩定", 0.4, [
                "檢查記憶體使用趨勢圖",
                "分析 heap dump",
                "增加容器記憶體限制",
            ]
        return "無法確定唯一的根本原因，建議進一步調查", 0.3, [
            "檢查最近的應用程式日誌",
            "比對部署時間與異常發生時間",
            "檢查外部服務的狀態",
            "考慮回滾至已知穩定的版本",
        ]


rca = RootCauseAnalyzer()
rca.add_code_change(CodeChange(
    file="src/services/order.py", author="Alice",
    commit_hash="a1b2c3", timestamp=datetime.now() - timedelta(hours=2),
```

## 13.3 效能調校

### 13.3.1 瓶頸識別

效能調校的第一步是準確識別系統的瓶頸所在。AI 可以透過分析效能資料、請求追蹤與資源使用模式，自動定位造成效能問題的元凶。

```python
"""
AI 效能瓶頸分析工具
分析請求追蹤與系統指標，識別效能瓶頸
"""
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
from collections import defaultdict
import statistics


@dataclass
class TraceSpan:
    name: str
    service: str
    start_time: float
    duration_ms: float
    span_type: str
    tags: Dict[str, str]


@dataclass
class PerformanceBottleneck:
    component: str
    metric: str
    current_value: float
    threshold: float
    impact: str
    suggestion: str
    severity: str


class BottleneckDetector:
    def __init__(self):
        self.thresholds = {
            "database_query": {"p95_ms": 500, "suggestion": "檢查資料庫索引與查詢效率"},
            "external_api": {"p95_ms": 2000, "suggestion": "考慮加入快取或非同步處理"},
            "cpu_bound": {"usage_percent": 80, "suggestion": "考慮平行化處理或升級硬體"},
            "memory_bound": {"usage_percent": 85, "suggestion": "檢查物件生命週期與記憶體管理"},
            "io_wait": {"percent": 20, "suggestion": "檢查磁碟效能，考慮使用 SSD"},
            "lock_contention": {"wait_ms": 100, "suggestion": "檢查鎖定策略，減少鎖定範圍"},
            "network_latency": {"ms": 100, "suggestion": "考慮將服務部署至相同區域"},
            "garbage_collection": {"pause_ms": 200, "suggestion": "調整 GC 參數或使用平行 GC"},
        }

    def analyze_traces(self, traces: List[TraceSpan]) -> List[PerformanceBottleneck]:
        bottlenecks = []
        service_breakdown = defaultdict(list)
        for trace in traces:
            service_breakdown[trace.service].append(trace)
        for service, service_traces in service_breakdown.items():
            total_time = sum(t.duration_ms for t in service_traces)
            by_type = defaultdict(list)
            for t in service_traces:
                by_type[t.span_type].append(t.duration_ms)
            for span_type, durations in by_type.items():
                if len(durations) > 3:
                    p95 = sorted(durations)[int(len(durations) * 0.95)]
                else:
                    p95 = max(durations)
                avg = statistics.mean(durations)
                total_span_time = sum(durations)
                pct_of_total = (total_span_time / total_time * 100) if total_time > 0 else 0
                if span_type in self.thresholds:
                    threshold_config = self.thresholds[span_type]
                    if "p95_ms" in threshold_config and p95 > threshold_config["p95_ms"]:
                        bottlenecks.append(PerformanceBottleneck(
                            component=f"{service}:{span_type}",
                            metric="p95_latency_ms",
                            current_value=p95,
                            threshold=threshold_config["p95_ms"],
                            impact=f"佔總時間 {pct_of_total:.1f}%",
                            suggestion=threshold_config["suggestion"],
                            severity="critical" if p95 > threshold_config["p95_ms"] * 2 else "warning",
                        ))
        bottlenecks.sort(key=lambda b: b.current_value / b.threshold if b.threshold else 0, reverse=True)
        return bottlenecks

    def analyze_system_metrics(self, metrics: Dict[str, float]) -> List[PerformanceBottleneck]:
        bottlenecks = []
        metric_mapping = {
            "cpu_usage": ("cpu_bound", "usage_percent"),
            "memory_usage": ("memory_bound", "usage_percent"),
            "io_wait": ("io_wait", "percent"),
            "gc_pause_ms": ("garbage_collection", "pause_ms"),
        }
        for metric_name, (threshold_key, threshold_field) in metric_mapping.items():
            if metric_name in metrics and threshold_key in self.thresholds:
                threshold_config = self.thresholds[threshold_key]
                threshold_value = threshold_config[threshold_field]
                if metrics[metric_name] > threshold_value:
                    bottlenecks.append(PerformanceBottleneck(
                        component=f"system:{metric_name}",
                        metric=metric_name,
                        current_value=metrics[metric_name],
                        threshold=threshold_value,
                        impact="系統整體效能受影響",
                        suggestion=threshold_config["suggestion"],
                        severity="critical" if metrics[metric_name] > threshold_value * 1.2 else "warning",
                    ))
        return bottlenecks

    def generate_performance_report(self, bottlenecks: List[PerformanceBottleneck]) -> str:
        report = ["=== 效能瓶頸分析報告 ==="]
        if not bottlenecks:
            report.append("目前未偵測到明顯的效能瓶頸。")
            return "\n".join(report)
        report.append(f"發現 {len(bottlenecks)} 個效能瓶頸:\n")
        for i, b in enumerate(bottlenecks, 1):
            report.append(f"瓶頸 #{i}: {b.component}")
            report.append(f"  指標: {b.metric}")
            report.append(f"  當前值: {b.current_value} (閾值: {b.threshold})")
            report.append(f"  影響: {b.impact}")
            report.append(f"  嚴重程度: {b.severity}")
            report.append(f"  建議: {b.suggestion}\n")
        return "\n".join(report)


detector = BottleneckDetector()

sample_traces = [
    TraceSpan("get_order", "order-service", 100.0, 450, "database_query", {"table": "orders"}),
    TraceSpan("get_order", "order-service", 100.0, 120, "database_query", {"table": "users"}),
    TraceSpan("get_order", "order-service", 100.5, 800, "external_api", {"api": "payment"}),
    TraceSpan("get_order", "order-service", 101.0, 250, "database_query", {"table": "products"}),
    TraceSpan("list_orders", "order-service", 200.0, 3000, "database_query", {"table": "orders"}),
    TraceSpan("list_orders", "order-service", 200.0, 3500, "database_query", {"table": "order_items"}),
]

bottlenecks = detector.analyze_traces(sample_traces)
system_metrics = {"cpu_usage": 92.5, "memory_usage": 78.3, "io_wait": 5.2, "gc_pause_ms": 350}
bottlenecks.extend(detector.analyze_system_metrics(system_metrics))
print(detector.generate_performance_report(bottlenecks))
```

### 13.3.2 查詢最佳化

資料庫查詢是 Web 應用程式最常見的效能瓶頸來源。AI 可以分析查詢執行計畫，自動建議索引策略與查詢重寫方案。

```python
"""
AI 查詢最佳化顧問
分析 SQL 查詢並提出最佳化建議
"""
import re
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass


@dataclass
class QueryOptimization:
    query: str
    issue: str
    severity: str
    impact: str
    suggestion: str
    estimated_improvement: str
    example: Optional[str] = None


class QueryOptimizer:
    def __init__(self):
        self.optimization_rules = [
            {
                "name": "select_star",
                "pattern": r"SELECT\s+\*\s+FROM",
                "severity": "medium",
                "suggestion": "只選取需要的欄位，減少資料傳輸量與記憶體使用",
                "example": "SELECT id, name, email FROM users  # 取代 SELECT * FROM users",
                "impact": "多餘的資料傳輸與記憶體開銷",
            },
            {
                "name": "no_limit",
                "pattern": r"SELECT\s+.+\s+FROM\s+\w+\s+(?!.*\bLIMIT\b)",
                "severity": "medium",
                "suggestion": "加上 LIMIT 限制回傳筆數，避免全表掃描",
                "example": "SELECT * FROM users WHERE status = 'active' LIMIT 100",
                "impact": "可能回傳大量資料導致應用程式記憶體不足",
            },
            {
                "name": "like_leading_wildcard",
                "pattern": r"LIKE\s+'%",
                "severity": "high",
                "suggestion": "避免 LIKE 查詢以萬用字元開頭，這會無法使用索引",
                "example": "使用全文檢索（Full-Text Search）取代 LIKE '%keyword%'",
                "impact": "即使有索引也無法使用，會觸發全表掃描",
            },
            {
                "name": "function_on_column",
                "pattern": r"(?:DATE|YEAR|MONTH|UPPER|LOWER|LENGTH|SUBSTR)\s*\(\s*(\w+)",
                "severity": "high",
                "suggestion": "避免在 WHERE 條件中對欄位套用函數，這會使索引失效",
                "example": (
                    "不建議: WHERE DATE(created_at) = '2026-01-01'\n"
                    "建議: WHERE created_at >= '2026-01-01' AND created_at < '2026-01-02'"
                ),
                "impact": "即使欄位有索引也無法使用",
            },
            {
                "name": "nested_subquery",
                "pattern": r"IN\s*\(\s*SELECT",
                "severity": "medium",
                "suggestion": "考慮使用 JOIN 或 EXISTS 替代 IN 子查詢",
                "example": (
                    "不建議: SELECT * FROM orders WHERE user_id IN (SELECT id FROM users WHERE status = 'active')\n"
                    "建議: SELECT o.* FROM orders o JOIN users u ON o.user_id = u.id WHERE u.status = 'active'"
                ),
                "impact": "子查詢可能導致逐行執行（correlated subquery）",
            },
        ]

    def analyze(self, query: str) -> List[QueryOptimization]:
        findings = []
        for rule in self.optimization_rules:
            if re.search(rule["pattern"], query, re.IGNORECASE):
                finding = QueryOptimization(
                    query=query,
                    issue=rule.get("suggestion", rule["name"]),
                    severity=rule["severity"],
                    impact=rule.get("impact", "查詢效能可能受到影響"),
                    suggestion=rule.get("suggestion", "請審視查詢並進行最佳化"),
                    estimated_improvement=self._estimate_improvement(rule["severity"]),
                    example=rule.get("example"),
                )
                findings.append(finding)
        join_tables = self._extract_join_tables(query)
        if join_tables:
            findings.append(QueryOptimization(
                query=query,
                issue=f"JOIN 查詢涉及多個表格（{', '.join(join_tables)}），請確保關聯欄位都有索引",
                severity="medium",
                impact="JOIN 操作若無索引可能導致大量資料掃描",
                suggestion="為 JOIN 條件中的外鍵欄位建立索引",
                estimated_improvement="10-100 倍",
                example=f"CREATE INDEX idx_{join_tables[0]}_{join_tables[1]}_fk ON {join_tables[0]}(foreign_key_id);",
            ))
        return findings

    def _extract_join_tables(self, query: str) -> List[str]:
        join_match = re.findall(r"JOIN\s+(\w+)", query, re.IGNORECASE)
        from_match = re.findall(r"FROM\s+(\w+)", query, re.IGNORECASE)
        return list(set(join_match + from_match))

    def _estimate_improvement(self, severity: str) -> str:
        improvements = {"high": "100-1000 倍", "medium": "10-100 倍", "low": "2-10 倍"}
        return improvements.get(severity, "未知")

    def batch_analyze(self, queries: List[str]) -> Dict[str, List[QueryOptimization]]:
        results = {}
        for i, query in enumerate(queries):
            label = f"查詢 #{i + 1}"
            findings = self.analyze(query)
            if findings:
                results[label] = findings
        return results


optimizer = QueryOptimizer()

sample_queries = [
    "SELECT * FROM orders",
    "SELECT * FROM users WHERE status = 'active'",
    "SELECT * FROM products WHERE name LIKE '%keyword%'",
    "SELECT o.*, u.name FROM orders o JOIN users u ON o.user_id = u.id",
    "SELECT DATE(created_at) as day, COUNT(*) FROM orders GROUP BY DATE(created_at)",
]

print("=== SQL 查詢最佳化建議 ===\n")
for label, findings in optimizer.batch_analyze(sample_queries).items():
    print(f"{label}:")
    for f in findings:
        print(f"  - [{f.severity}] {f.issue}")
        print(f"    建議: {f.suggestion}")
        print(f"    預估改善: {f.estimated_improvement}")
        if f.example:
            print(f"    範例:\n      {f.example.replace(chr(10), chr(10) + '      ')}")
    print()
```

### 13.3.3 資源使用分析

AI 可以分析應用程式的資源使用模式，識別記憶體洩漏、CPU 熱點與 I/O 瓶頸，並提供具體的改善方向。

```python
"""
資源使用分析器
監控與分析應用程式的資源使用模式
"""
from typing import List, Dict, Optional
from dataclasses import dataclass
import statistics
import time


@dataclass
class ResourceSample:
    timestamp: float
    cpu_percent: float
    memory_mb: float
    disk_io_mbps: float
    network_io_mbps: float
    thread_count: int
    open_fds: int


@dataclass
class ResourceAnalysis:
    resource_type: str
    trend: str
    current_value: float
    baseline_value: float
    deviation_percent: float
    recommendation: str
    severity: str


class ResourceAnalyzer:
    def __init__(self, window_size: int = 100):
        self.window_size = window_size
        self.samples: List[ResourceSample] = []

    def add_sample(self, sample: ResourceSample):
        self.samples.append(sample)
        if len(self.samples) > self.window_size:
            self.samples = self.samples[-self.window_size:]

    def analyze_memory(self) -> Optional[ResourceAnalysis]:
        if len(self.samples) < 10:
            return None
        recent = self.samples[-10:]
        older = self.samples[:-10]
        current_avg = statistics.mean(s.memory_mb for m in recent)
        if older:
            baseline_avg = statistics.mean(m.memory_mb for m in older)
            deviation = ((current_avg - baseline_avg) / baseline_avg) * 100
            if deviation > 20:
                return ResourceAnalysis(
                    resource_type="memory",
                    trend="increasing",
                    current_value=current_avg,
                    baseline_value=baseline_avg,
                    deviation_percent=round(deviation, 1),
                    recommendation=(
                        "記憶體使用量持續上升，可能存在記憶體洩漏:\n"
                        "1. 使用記憶體分析器（如 memory-profiler）定位記憶體消耗熱點\n"
                        "2. 檢查全域快取與靜態集合是否無限制增長\n"
                        "3. 確認物件是否正確實現 __del__ 或 dispose 模式\n"
                        "4. 考慮使用弱參考（weakref）儲存快取資料"
                    ),
                    severity="critical" if deviation > 50 else "warning",
                )
        return None

    def analyze_cpu(self) -> Optional[ResourceAnalysis]:
        if len(self.samples) < 10:
            return None
        recent = self.samples[-10:]
        current_avg = statistics.mean(s.cpu_percent for s in recent)
        if current_avg > 70:
            return ResourceAnalysis(
                resource_type="cpu",
                trend="high_utilization",
                current_value=current_avg,
                baseline_value=0,
                deviation_percent=0,
                recommendation=(
                    f"CPU 使用率平均 {current_avg:.1f}%:\n"
                    "1. 使用 cProfile 或 py-spy 找出 CPU 熱點\n"
                    "2. 檢查是否有無限迴圈或密集計算\n"
                    "3. 考慮使用非同步處理或增加 Worker 數量\n"
                    "4. 確認是否有不必要的輪詢（polling）機制"
                ),
                severity="critical" if current_avg > 90 else "warning",
            )
        return None

    def analyze_threads(self) -> Optional[ResourceAnalysis]:
        if len(self.samples) < 10:
            return None
        recent = self.samples[-10:]
        thread_counts = [s.thread_count for s in recent]
        current = statistics.mean(thread_counts)
        trend = "increasing" if thread_counts[-1] > thread_counts[0] * 1.2 else "stable"
        if current > 100 or trend == "increasing":
            return ResourceAnalysis(
                resource_type="threads",
                trend=trend,
                current_value=current,
                baseline_value=thread_counts[0],
                deviation_percent=((current - thread_counts[0]) / thread_counts[0]) * 100 if thread_counts[0] > 0 else 0,
                recommendation=(
                    f"執行緒數量 {current:.0f}:\n"
                    "1. 檢查是否正確使用執行緒池（ThreadPoolExecutor）\n"
                    "2. 確認已建立的執行緒有正確關閉\n"
                    "3. 考慮使用非同步 IO 取代多執行緒\n"
                    "4. 設定合理的執行緒池大小上限"
                ),
                severity="critical" if current > 200 else "warning",
            )
        return None

    def generate_report(self) -> str:
        analyses = [
            self.analyze_memory(),
            self.analyze_cpu(),
            self.analyze_threads(),
        ]
        report = ["=== 資源使用分析報告 ==="]
        has_issues = False
        for analysis in analyses:
```

## 13.4 技術債管理

### 13.4.1 識別與量化

技術債（Technical Debt）是軟體維護中難以避免的問題。AI 可以透過靜態分析、程式碼複雜度評估與模式識別，自動識別技術債並量化其影響。

```python
"""
技術債識別與量化工具
分析程式碼品質指標並量化技術債
"""
import re
import math
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
from pathlib import Path


@dataclass
class TechnicalDebtItem:
    file: str
    line: int
    category: str
    description: str
    estimated_hours: float
    impact: str
    priority: str


class TechnicalDebtAnalyzer:
    def __init__(self):
        self.debt_patterns = [
            {
                "category": "duplicate_code",
                "pattern": None,
                "description": "重複的程式碼區塊",
                "base_hours": 2.0,
            },
            {
                "category": "complex_function",
                "pattern": r"def\s+\w+",
                "description": "函式複雜度過高",
                "base_hours": 3.0,
            },
            {
                "category": "long_function",
                "pattern": r"def\s+\w+",
                "description": "函式行數過長",
                "base_hours": 2.0,
            },
            {
                "category": "magic_number",
                "pattern": r"[^_a-zA-Z]\d{4,}[^_a-zA-Z]",
                "description": "使用了魔術數字",
                "base_hours": 0.5,
            },
            {
                "category": "hardcoded_config",
                "pattern": r"(?:password|secret|api_key|token)\s*[:=]\s*['\"][^'\"]+['\"]",
                "description": "硬編碼的敏感資訊",
                "base_hours": 1.0,
            },
            {
                "category": "todo_comment",
                "pattern": r"#\s*(?:TODO|FIXME|HACK|XXX|TEMP)",
                "description": "未處理的待辦事項",
                "base_hours": 1.0,
            },
            {
                "category": "exception_silencing",
                "pattern": r"except\s*:\s*pass|except\s+Exception\s*:\s*pass",
                "description": "靜默吞沒的例外",
                "base_hours": 1.5,
            },
            {
                "category": "deeply_nested",
                "pattern": r"^\s{20,}(?:if|for|while|with|try)",
                "description": "巢狀結構過深",
                "base_hours": 2.5,
            },

    def analyze_file(self, filepath: Path) -> List[TechnicalDebtItem]:
        items = []
        try:
            content = filepath.read_text(encoding="utf-8")
        except (IOError, UnicodeDecodeError):
            return items
        lines = content.split("\n")
        for i, line in enumerate(lines, 1):
            for pattern in self.debt_patterns:
                if pattern["pattern"] and re.search(pattern["pattern"], line):
                    items.append(TechnicalDebtItem(
                        file=str(filepath),
                        line=i,
                        category=pattern["category"],
                        description=pattern["description"],
                        estimated_hours=pattern["base_hours"],
                        impact=self._estimate_impact(pattern["base_hours"]),
                        priority=self._assign_priority(pattern["base_hours"]),
                    ))
        long_funcs = self._find_long_functions(lines)
        items.extend(long_funcs)
        comment_blocks = self._find_todo_comments(lines)
        items.extend(comment_blocks)
        return items

    def _find_long_functions(self, lines: List[str]) -> List[TechnicalDebtItem]:
        items = []
        func_start = None
        func_name = None
        brace_count = 0
        for i, line in enumerate(lines, 1):
            func_match = re.match(r"^(\s*)def\s+(\w+)\s*\(", line)
            if func_match:
                if func_start is not None and (i - func_start) > 50:
                    items.append(TechnicalDebtItem(
                        file="", line=func_start, category="long_function",
                        description=f"函式 '{func_name}' 長達 {i - func_start} 行",
                        estimated_hours=2.0,
                        impact="可讀性與可維護性下降",
                        priority="medium",
                    ))
                func_start = i
                func_name = func_match.group(2)
                brace_count = 0
            if "{" in line:
                brace_count += line.count("{")
            if "}" in line:
                brace_count -= line.count("}")
                if brace_count <= 0 and func_start is not None and (i - func_start) > 50:
                    items.append(TechnicalDebtItem(
                        file="", line=func_start, category="long_function",
                        description=f"函式 '{func_name}' 長達 {i - func_start} 行",
                        estimated_hours=2.0,
                        impact="可讀性與可維護性下降",
                        priority="medium",
                    ))
                    func_start = None
        return items

    def _find_todo_comments(self, lines: List[str]) -> List[TechnicalDebtItem]:
        items = []
        for i, line in enumerate(lines, 1):
            match = re.search(r"#\s*(TODO|FIXME|HACK|XXX|TEMP)\b\s*(.*)", line)
            if match:
                tag = match.group(1)
                desc = match.group(2).strip()
                items.append(TechnicalDebtItem(
                    file="", line=i, category="todo_comment",
                    description=f"{tag}: {desc}",
                    estimated_hours=1.0,
                    impact="可能存在未完成的實作或已知問題",
                    priority="low" if tag == "TODO" else "medium",
                ))
        return items

    def _estimate_impact(self, hours: float) -> str:
        if hours >= 3:
            return "高"
        elif hours >= 1.5:
            return "中"
        return "低"

    def _assign_priority(self, hours: float) -> str:
        if hours >= 3:
            return "high"
        elif hours >= 1.5:
            return "medium"
        return "low"

    def calculate_debt_ratio(self, items: List[TechnicalDebtItem], total_lines: int) -> float:
        total_hours = sum(item.estimated_hours for item in items)
        if total_lines == 0:
            return 0.0
        return total_hours / (total_lines / 100)

    def generate_debt_report(self, items: List[TechnicalDebtItem]) -> Dict:
        category_summary = {}
        for item in items:
            if item.category not in category_summary:
                category_summary[item.category] = {"count": 0, "hours": 0.0}
            category_summary[item.category]["count"] += 1
            category_summary[item.category]["hours"] += item.estimated_hours
        total_hours = sum(item.estimated_hours for item in items)
        priority_counts = {"high": 0, "medium": 0, "low": 0}
        for item in items:
            priority_counts[item.priority] = priority_counts.get(item.priority, 0) + 1
        return {
            "total_items": len(items),
            "total_estimated_hours": round(total_hours, 1),
            "priority_breakdown": priority_counts,
            "category_breakdown": category_summary,
            "items": items,
        }
```

### 13.4.3 自動化重構

AI 可以自動執行某些標準化的重構操作，例如重新命名變數、提取函式、消除重複程式碼等。

```python
"""
自動化重構工具
安全地執行常見的重構操作
"""
import ast
import re
from typing import List, Dict, Optional


class AutoRefactor:
    def __init__(self):
        self.refactorings = []

    def rename_variable(self, source: str, old_name: str, new_name: str) -> str:
        tree = ast.parse(source)
        renamer = VariableRenamer(old_name, new_name)
        renamer.visit(tree)
        return ast.unparse(tree)

    def extract_function(self, source: str, start_line: int, end_line: int, new_func_name: str) -> str:
        lines = source.split("\n")
        extracted_lines = lines[start_line - 1:end_line]
        indent = len(extracted_lines[0]) - len(extracted_lines[0].lstrip())
        min_indent = min(len(l) - len(l.lstrip()) for l in extracted_lines if l.strip())
        dedented = [l[min_indent:] for l in extracted_lines]
        lines[start_line - 1:end_line] = [
            " " * indent + f"def {new_func_name}():",
            *["    " + l for l in dedented],
        ]
        return "\n".join(lines)

    def remove_dead_code(self, source: str) -> str:
        unused_vars = self._find_unused_variables(source)
        lines = source.split("\n")
        result = []
        for i, line in enumerate(lines):
            stripped = line.strip()
            if any(f"{var} =" in stripped and stripped.startswith("#") == False for var in unused_vars):
                continue
            result.append(line)
        return "\n".join(result)

    def _find_unused_variables(self, source: str) -> List[str]:
        try:
            tree = ast.parse(source)
            finder = UnusedVariableFinder()
            finder.visit(tree)
            return finder.unused_vars
        except SyntaxError:
            return []


class VariableRenamer(ast.NodeTransformer):
    def __init__(self, old_name: str, new_name: str):
        self.old_name = old_name
        self.new_name = new_name

    def visit_Name(self, node: ast.Name) -> ast.Name:
        if node.id == self.old_name:
            node.id = self.new_name
        return node


class UnusedVariableFinder(ast.NodeVisitor):
    def __init__(self):
        self.defined = {}
        self.used = set()

    def visit_Assign(self, node: ast.Assign):
        for target in node.targets:
            if isinstance(target, ast.Name):
                self.defined[target.id] = node.lineno
        self.generic_visit(node)

    def visit_Name(self, node: ast.Name):
        if isinstance(node.ctx, ast.Load):
            self.used.add(node.id)


refactor = AutoRefactor()
sample = """
def process():
    x = 10
    y = 20
    result = x + y
    old_name = 100
    return result
"""
print("=== 自動化重構範例 ===")
print("原始程式碼:")
print(sample)
renamed = refactor.rename_variable(sample, "old_name", "new_name")
print("重新命名後:")
print(renamed)
```

## 13.5 異常偵測與預測性維護

### 13.5.1 日誌異常模式

AI 可以從大量日誌資料中學習正常的運作模式，並在偵測到異常行為時即時告警。以下是一個基於統計與規則的異常偵測系統：

```python
"""
異常偵測系統
基於日誌分析與統計模型的異常偵測
"""
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
from datetime import datetime, timedelta
from collections import defaultdict, Counter
import statistics
import re


@dataclass
class LogEvent:
    timestamp: datetime
    level: str
    service: str
    message: str
    duration_ms: Optional[float] = None


@dataclass
class AnomalyAlert:
    event_type: str
    severity: str
    description: str
    detected_at: datetime
    affected_service: str
    metrics: Dict
    suggested_action: str


class AnomalyDetector:
    def __init__(self):
        self.events: List[LogEvent] = []
        self.baselines: Dict[str, Dict] = {}
        self.baseline_window = timedelta(hours=24)

    def ingest_event(self, event: LogEvent) -> Optional[AnomalyAlert]:
        self.events.append(event)
        self.events = self.events[-10000:]
        alerts = []
        level_alert = self._check_error_rate(event.service)
        if level_alert:
            alerts.append(level_alert)
        duration_alert = self._check_duration_anomaly(event)
        if duration_alert:
            alerts.append(duration_alert)
        pattern_alert = self._check_known_patterns(event)
        if pattern_alert:
            alerts.append(pattern_alert)
        return alerts[0] if alerts else None

    def _check_error_rate(self, service: str) -> Optional[AnomalyAlert]:
        recent = [e for e in self.events if e.service == service and e.timestamp > datetime.now() - timedelta(minutes=5)]
        if len(recent) < 10:
            return None
        error_count = sum(1 for e in recent if e.level in ("ERROR", "CRITICAL"))
        total_count = len(recent)
        error_rate = (error_count / total_count) * 100
        historical = [e for e in self.events if e.service == service and e.timestamp > datetime.now() - self.baseline_window and e.timestamp <= datetime.now() - timedelta(minutes=5)]
        if historical:
            hist_errors = sum(1 for e in historical if e.level in ("ERROR", "CRITICAL"))
            hist_total = len(historical)
            hist_error_rate = (hist_errors / hist_total) * 100 if hist_total > 0 else 0
            if error_rate > hist_error_rate * 3 and error_rate > 5:
                return AnomalyAlert(
                    event_type="error_rate_spike",
                    severity="critical",
                    description=f"服務 {service} 錯誤率異常: {error_rate:.1f}%（基線: {hist_error_rate:.1f}%）",
                    detected_at=datetime.now(),
                    affected_service=service,
                    metrics={"current_error_rate": error_rate, "baseline_error_rate": hist_error_rate},
                    suggested_action="檢查最近的部署變更，查看錯誤日誌找出異常原因",
                )
        return None

    def _check_duration_anomaly(self, event: LogEvent) -> Optional[AnomalyAlert]:
        if event.duration_ms is None:
            return None
        recent_durations = [
            e.duration_ms for e in self.events
            if e.service == event.service and e.duration_ms is not None
            and e.timestamp > datetime.now() - timedelta(hours=1)
        ]
        if len(recent_durations) < 20:
            return None
        mean = statistics.mean(recent_durations)
        stdev = statistics.stdev(recent_durations) if len(recent_durations) > 1 else 0
        if stdev > 0 and event.duration_ms > mean + 3 * stdev:
            return AnomalyAlert(
                event_type="latency_spike",
                severity="warning",
                description=f"服務 {event.service} 請求延遲異常: {event.duration_ms:.0f}ms（平均: {mean:.0f}ms）",
                detected_at=datetime.now(),
                affected_service=event.service,
                metrics={"current_ms": event.duration_ms, "mean_ms": mean, "stdev_ms": stdev},
                suggested_action="檢查資料庫查詢效能與外部 API 回應時間",
            )
        return None

    def _check_known_patterns(self, event: LogEvent) -> Optional[AnomalyAlert]:
        anomaly_patterns = [
            (r"OOM|OutOfMemory|Cannot allocate memory", "out_of_memory", "critical",
             "應用程式記憶體不足，請檢查記憶體使用狀況並考慮增加資源"),
            (r"deadlock detected|deadlock", "deadlock", "critical",
             "資料庫發生死結，請檢查交易順序與鎖定策略"),
            (r"circuit breaker (?:open|half-open)", "circuit_breaker", "warning",
             "斷路器開啟，下游服務可能不穩定，請檢查外部依賴"),
            (r"disk full|No space left", "disk_full", "critical",
             "磁碟空間不足，請清理日誌或擴充儲存空間"),
            (r"too many open files", "fd_leak", "warning",
             "檔案描述子耗盡，請檢查資源釋放邏輯"),
        ]
        for pattern, event_type, severity, action in anomaly_patterns:
            if re.search(pattern, event.message, re.IGNORECASE):
                return AnomalyAlert(
                    event_type=event_type,
                    severity=severity,
                    description=f"服務 {event.service} 觸發已知異常模式: {event.message[:100]}",
                    detected_at=datetime.now(),
                    affected_service=event.service,
                    metrics={},
                    suggested_action=action,
                )
        return None

    def generate_daily_report(self) -> Dict:
        today = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
        today_events = [e for e in self.events if e.timestamp >= today]
        error_counts = Counter(e.service for e in today_events if e.level in ("ERROR", "CRITICAL"))
        level_counts = Counter(e.level for e in today_events)
        return {
            "date": today.date().isoformat(),
            "total_events": len(today_events),
            "level_breakdown": dict(level_counts),
            "top_error_services": dict(error_counts.most_common(5)),
            "total_anomalies": 0,
        }


detector = AnomalyDetector()

    LogEvent(now - timedelta(seconds=25), "ERROR", "order-service", "connection timeout after 5100ms", 5100),
    LogEvent(now - timedelta(seconds=20), "ERROR", "order-service", "connection timeout after 4900ms", 4900),
```

### 13.5.2 效能衰退預警

AI 可以持續監控效能指標，並在偵測到衰退趨勢時提前發出預警，讓團隊在使用者感受到問題之前就採取行動。

```python
"""
效能衰退預警系統
監控效能指標的長期趨勢，提前預警效能衰退
"""
from typing import List, Dict, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta
import statistics


@dataclass
class PerformanceDataPoint:
    timestamp: datetime
    metric_name: str
    value: float
    service: str


class PerformanceRegressionDetector:
    def __init__(self, window_days: int = 7):
        self.window_days = window_days
        self.data: List[PerformanceDataPoint] = []

    def add_data_point(self, point: PerformanceDataPoint):
        self.data.append(point)
        cutoff = datetime.now() - timedelta(days=self.window_days)
        self.data = [d for d in self.data if d.timestamp > cutoff]

    def detect_regression(self, metric_name: str, service: str, threshold_pct: float = 10.0) -> Optional[Dict]:
        points = [
            d for d in self.data
            if d.metric_name == metric_name and d.service == service
        ]
        if len(points) < 10:
            return None
        midpoint = len(points) // 2
        first_half = [p.value for p in points[:midpoint]]
        second_half = [p.value for p in points[midpoint:]]
        avg_first = statistics.mean(first_half)
        avg_second = statistics.mean(second_half)
        if avg_first > 0:
            change_pct = ((avg_second - avg_first) / avg_first) * 100
            if change_pct > threshold_pct:
                return {
                    "metric": metric_name,
                    "service": service,
                    "change_percent": round(change_pct, 1),
                    "previous_avg": round(avg_first, 2),
                    "current_avg": round(avg_second, 2),
                    "severity": "critical" if change_pct > threshold_pct * 2 else "warning",
                    "detected_at": datetime.now(),
                }
        return None


detector = PerformanceRegressionDetector()
base_time = datetime.now()

    degradation = i / 200 * 15
    detector.add_data_point(PerformanceDataPoint(
        timestamp=base_time - timedelta(hours=i),
        metric_name="p95_latency_ms",
        value=200 + degradation,
        service="api-gateway",
    ))

    print(f"當前平均值: {result['current_avg']}")
```

```python
"""
安全漏洞預測工具
分析依賴項目的安全風險與版本趨勢
"""
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass
from datetime import datetime
import re


@dataclass
class DependencyInfo:
    name: str
    current_version: str
    latest_version: str
    release_dates: List[str]
    has_known_vulnerabilities: bool
    vulnerability_count: int
    maintainer_activity: str


class VulnerabilityPredictor:
    def __init__(self):
        self.risk_factors = [
            ("version_age", 0.3),
            ("maintainer_inactivity", 0.25),
            ("known_vulnerabilities", 0.3),
            ("dependency_depth", 0.15),
        ]

    def predict_risk(self, dep: DependencyInfo) -> Dict:
        risk_score = 0.0
        details = []
        version_age = self._calculate_version_age(dep)
        if version_age > 365:
            risk_score += 0.3
            details.append(f"版本已發布超過 {version_age} 天，可能存在未修復的漏洞")
        if dep.maintainer_activity == "low":
            risk_score += 0.25
            details.append("維護者活躍度低，漏洞修復可能延遲")
        if dep.has_known_vulnerabilities:
            risk_score += 0.3
            details.append(f"已知 {dep.vulnerability_count} 個安全漏洞")
        if risk_score > 0.5:
            return {
                "dependency": dep.name,
                "current_version": dep.current_version,
                "risk_score": round(risk_score, 2),
                "risk_level": "high" if risk_score > 0.7 else "medium",
                "details": details,
                "recommendation": f"建議升級至 {dep.latest_version}",
                "deadline": "立即處理" if risk_score > 0.7 else "下個衝刺處理",
            }
        return {
            "dependency": dep.name,
            "risk_score": round(risk_score, 2),
            "risk_level": "low",
            "details": ["風險較低，但仍需持續關注安全公告"],
        }

    def _calculate_version_age(self, dep: DependencyInfo) -> int:
        if dep.release_dates:
            try:
                last_release = max(dep.release_dates)
                release_date = datetime.fromisoformat(last_release)
                return (datetime.now() - release_date).days
            except (ValueError, TypeError):
                pass
        return 0
```

### 13.6.1 套件更新建議

AI 可以監控專案的依賴套件，根據版本變更、安全公告與相容性分析，提供精確的更新建議。

```python
"""
AI 依賴管理工具
分析套件依賴並提供更新建議
"""
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass


@dataclass
class PackageInfo:
    name: str
    current_version: str
    latest_version: str
    update_type: str
    changelog_summary: str
    compatibility_risk: str
    security_fixes: bool


class DependencyManager:
    def __init__(self):
        self.package_db: Dict[str, Dict] = {}

    def analyze_requirements(self, requirements: List[str]) -> List[PackageInfo]:
        results = []
        for req in requirements:
            parts = req.strip().split("==")
            name = parts[0]
            current = parts[1] if len(parts) > 1 else "unknown"
            info = self.package_db.get(name, {})
            latest = info.get("latest", current)
            update_type = self._determine_update_type(current, latest)
            results.append(PackageInfo(
                name=name,
                current_version=current,
                latest_version=latest,
                update_type=update_type,
                changelog_summary=info.get("changelog", ""),
                compatibility_risk=self._assess_risk(update_type),
                security_fixes=info.get("security", False),
            ))
        return results

    def _determine_update_type(self, current: str, latest: str) -> str:
        if current == latest:
            return "up_to_date"
        try:
            cur_parts = [int(p) for p in current.split(".")]
            lat_parts = [int(p) for p in latest.split(".")]
            if lat_parts[0] > cur_parts[0]:
                return "major"
            elif lat_parts[1] > cur_parts[1]:
                return "minor"
            elif lat_parts[2] > cur_parts[2]:
                return "patch"
        except (ValueError, IndexError):
            return "unknown"
        return "up_to_date"

    def _assess_risk(self, update_type: str) -> str:
        risks = {"major": "high", "minor": "medium", "patch": "low", "up_to_date": "none"}
        return risks.get(update_type, "unknown")

    def generate_update_plan(self, packages: List[PackageInfo]) -> str:
        plan = ["=== 套件更新計畫 ==="]
        security_updates = [p for p in packages if p.security_fixes]
        if security_updates:
            plan.append(f"\n安全更新（{len(security_updates)} 項）:")
            for p in security_updates:
                plan.append(f"  - {p.name}: {p.current_version} -> {p.latest_version}")
        patch_updates = [p for p in packages if p.update_type == "patch" and not p.security_fixes]
        if patch_updates:
            plan.append(f"\n修補更新（{len(patch_updates)} 項）:")
            for p in patch_updates:
                plan.append(f"  - {p.name}: {p.current_version} -> {p.latest_version}（風險: {p.compatibility_risk}）")
        minor_updates = [p for p in packages if p.update_type == "minor"]
        if minor_updates:
            plan.append(f"\n次要更新（{len(minor_updates)} 項）:")
            for p in minor_updates:
                plan.append(f"  - {p.name}: {p.current_version} -> {p.latest_version}（風險: {p.compatibility_risk}）")
                plan.append(f"    變更摘要: {p.changelog_summary[:80]}...")
        major_updates = [p for p in packages if p.update_type == "major"]
        if major_updates:
            plan.append(f"\n主要更新（{len(major_updates)} 項，需要謹慎評估）:")
            for p in major_updates:
                plan.append(f"  - {p.name}: {p.current_version} -> {p.latest_version}（風險: {p.compatibility_risk}）")
                plan.append(f"    變更摘要: {p.changelog_summary[:80]}...")
        return "\n".join(plan)


manager = DependencyManager()
manager.package_db = {
    "requests": {"latest": "2.32.0", "changelog": "修正 HTTP/2 連線問題，安全性更新", "security": True},
    "flask": {"latest": "3.1.0", "changelog": "新增非同步支援，棄用部分舊 API", "security": False},
    "sqlalchemy": {"latest": "2.0.35", "changelog": "修正 ORM 延遲載入問題", "security": False},
}

requirements = ["requests==2.31.0", "flask==3.0.0", "sqlalchemy==2.0.30"]
packages = manager.analyze_requirements(requirements)
print(manager.generate_update_plan(packages))
```

### 13.6.2 相容性分析

AI 可以分析套件的相依關係，預測升級可能導致的相容性問題，並提出具體的解決方案。

```python
"""
相容性分析工具
預測套件升級的相容性風險
"""
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass


@dataclass
class CompatibilityIssue:
    package: str
    version_change: str
    issue_type: str
    description: str
    affected_modules: List[str]
    suggested_fix: str
    estimated_effort: str


class CompatibilityAnalyzer:
    def __init__(self):
        self.known_breaking_changes = {
            "flask": {
                "3.0": [
                    {
                        "type": "api_removal",
                        "description": "移除 flask.Flash 類別",
                        "modules": ["flask.Flash", "flask.flash"],
                        "fix": "改用 flask.flash() 函式替代",
                        "effort": "低",
                    },
                    {
                        "type": "signature_change",
                        "description": "app.run() 不再接受 debug 參數",
                        "modules": ["flask.app.run"],
                        "fix": "使用 app.config['DEBUG'] = True 或傳入環境變數",
                        "effort": "低",
                    },
                ],
            },
            "sqlalchemy": {
                "2.0": [
                    {
                        "type": "api_removal",
                        "description": "移除 query 屬性，改用 select() 函式",
                        "modules": ["sqlalchemy.orm.Query"],
                        "fix": "將 Session.query(Model) 改為 select(Model)",
                        "effort": "中",
                    },
                    {
                        "type": "behavior_change",
                        "description": "預設改為延遲載入（lazy loading）",
                        "modules": ["sqlalchemy.orm.relationship"],
                        "fix": "在 relationship 上明確指定 lazy='select' 或 lazy='joined'",
                        "effort": "低",
                    },
                ],
            },
        }

    def analyze_upgrade(self, package: str, from_version: str, to_version: str) -> List[CompatibilityIssue]:
        issues = []
        if package in self.known_breaking_changes:
            for version, changes in self.known_breaking_changes[package].items():
                if self._version_compare(from_version, version) < 0 and self._version_compare(to_version, version) >= 0:
                    for change in changes:
                        issues.append(CompatibilityIssue(
                            package=package,
                            version_change=f"{from_version} -> {to_version}",
                            issue_type=change["type"],
                            description=change["description"],
                            affected_modules=change["modules"],
                            suggested_fix=change["fix"],
                            estimated_effort=change["effort"],
                        ))
        return issues

    def _version_compare(self, v1: str, v2: str) -> int:
        try:
            p1 = [int(x) for x in v1.split(".")]
            p2 = [int(x) for x in v2.split(".")]
            for a, b in zip(p1, p2):
                if a != b:
                    return -1 if a < b else 1
            return 0
        except (ValueError, IndexError):
            return 0


analyzer = CompatibilityAnalyzer()
issues = analyzer.analyze_upgrade("flask", "2.3.0", "3.1.0")
print("=== 相容性分析 ===")
for issue in issues:
    print(f"套件: {issue.package} ({issue.version_change})")
    print(f"類型: {issue.issue_type}")
    print(f"說明: {issue.description}")
    print(f"影響模組: {', '.join(issue.affected_modules)}")
    print(f"修復建議: {issue.suggested_fix}")
    print(f"預估工時: {issue.estimated_effort}\n")
```
